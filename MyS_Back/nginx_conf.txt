worker_processes auto;  # Detecta automáticamente los núcleos del CPU

events { 
    worker_connections 1024; 
}

http {
    include mime.types;
    default_type application/octet-stream;
    sendfile on;
    keepalive_timeout 65;
    
    # Optimizaciones para producción
    tcp_nopush on;
    tcp_nodelay on;
    types_hash_max_size 2048;
    
    # Compresión gzip
    gzip on;
    gzip_vary on;
    gzip_proxied any;
    gzip_comp_level 6;
    gzip_types text/plain text/css text/xml text/javascript 
               application/json application/javascript application/xml+rss 
               application/rss+xml font/truetype font/opentype 
               application/vnd.ms-fontobject image/svg+xml;

    # ==============================
    #  Servidor principal HTTP
    # ==============================
    server {
        listen 80;
        server_name mynetspace.ddns.net localhost;

        # LOGS (cambia a 'warn' en producción para menos verbosidad)
        access_log /var/log/nginx/myspace_access.log;
        error_log /var/log/nginx/myspace_error.log warn;
        
        # HEADERS DE SEGURIDAD
        add_header X-Frame-Options "SAMEORIGIN" always;
        add_header X-Content-Type-Options "nosniff" always;
        add_header X-XSS-Protection "1; mode=block" always;

        # Tamaño máximo de upload (ajusta según necesites)
        client_max_body_size 10M;

        # =========================
        #  Django API
        # =========================
        location /api/ {
            proxy_pass http://127.0.0.1:8001;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            
            # Timeouts para evitar conexiones colgadas
            proxy_connect_timeout 60s;
            proxy_send_timeout 60s;
            proxy_read_timeout 60s;
            
            # Manejo de preflight OPTIONS
            if ($request_method = 'OPTIONS') {
                add_header 'Access-Control-Allow-Origin' '$http_origin' always;
                add_header 'Access-Control-Allow-Methods' 'GET, POST, PUT, DELETE, PATCH, OPTIONS' always;
                add_header 'Access-Control-Allow-Headers' 'Authorization, Content-Type, X-CSRFToken' always;
                add_header 'Access-Control-Allow-Credentials' 'true' always;
                add_header 'Access-Control-Max-Age' 1728000;
                add_header 'Content-Type' 'text/plain; charset=utf-8';
                add_header 'Content-Length' 0;
                return 204;
            }
        }

        # =========================
        #  Django admin
        # =========================
        location /admin/ {
            proxy_pass http://127.0.0.1:8001/admin/;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }

        # =========================
        #  Django static files
        # =========================
        location /django_static/ {
            alias /MySpace/MyS_Back/staticfiles/;
            expires 1y;
            add_header Cache-Control "public, immutable";
            access_log off;
        }

        # =========================
        #  React static files (JS, CSS)
        # =========================
        location /static/ {
            alias /MySpace/MyS_Front/build/static/;
            expires 1y;
            add_header Cache-Control "public, immutable";
            access_log off;
        }

        # =========================
        #  React manifest, favicon, logos
        # =========================
        location ~* ^/(manifest\.json|favicon\.ico|logo192\.png|logo512\.png|robots\.txt)$ {
            root /MySpace/MyS_Front/build;
            expires 30d;
            add_header Cache-Control "public, immutable";
            access_log off;
        }

        # =========================
        #  React index.html
        # =========================
        location / {
            root /MySpace/MyS_Front/build;
            index index.html;
            try_files $uri /index.html;
            
            # No cachear el index.html
            add_header Cache-Control "no-cache, no-store, must-revalidate";
            add_header Pragma "no-cache";
            add_header Expires "0";
        }

        # =========================
        #  Páginas de error
        # =========================
        error_page 500 502 503 504 /50x.html;
        location = /50x.html {
            root /usr/share/nginx/html;
        }
    }
}
