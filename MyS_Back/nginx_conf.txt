# MySpace\MyS_Back\nginx_conf.txt
# se modifica nginx.conf para que pueda actuar como proxy entre Django y React

worker_processes 1;

events { 
    worker_connections 1024; 
}

http {
    include mime.types;
    default_type application/octet-stream;
    sendfile on;
    keepalive_timeout 65;

    # ==============================
    #  Redirigir HTTP → HTTPS
    # ==============================
    server {
        listen 80;
        server_name localhost;
        return 301 https://$host$request_uri;
    }

    # ==============================
    #  Servidor principal HTTPS (DESARROLLO)
    # ==============================
    server {
        listen 443 ssl;
        server_name localhost;

        # --- Certificados generados con mkcert ---
        ssl_certificate     C:/certs/localhost.pem;
        ssl_certificate_key C:/certs/localhost-key.pem;
        ssl_protocols TLSv1.2 TLSv1.3;
        ssl_ciphers HIGH:!aNULL:!MD5;

        # LOGS DETALLADOS PARA DESARROLLO
        access_log logs/access.log;
        error_log logs/error.log debug;  # Nivel debug para desarrollo
        
        # HEADERS DE DESARROLLO (mostrar errores)
        add_header X-Debug-Mode "Development" always;
        add_header X-Frame-Options "SAMEORIGIN" always;

        # =========================
        #  Django API
        # =========================
        location /api/ {
            proxy_pass http://127.0.0.1:8001;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            
            # ELIMINAR CORS HEADERS (Django los maneja)
            # Solo manejar preflight de OPTIONS
            if ($request_method = 'OPTIONS') {
                add_header 'Access-Control-Allow-Origin' '$http_origin' always;
                add_header 'Access-Control-Allow-Methods' 'GET, POST, PUT, DELETE, PATCH, OPTIONS' always;
                add_header 'Access-Control-Allow-Headers' 'Authorization, Content-Type, X-CSRFToken' always;
                add_header 'Access-Control-Allow-Credentials' 'true' always;
                add_header 'Access-Control-Max-Age' 1728000;
                add_header 'Content-Type' 'text/plain; charset=utf-8';
                add_header 'Content-Length' 0;
                return 204;
            }
        }

        # =========================
        #  Django admin
        # =========================
        location /admin/ {
            proxy_pass http://127.0.0.1:8001/admin/;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }

        # =========================
        #  Django static files
        # =========================
        location /django_static/ {
            alias C:/MyS_U4/MySpace/MyS_Back/staticfiles/;
            expires 1y;
            add_header Cache-Control "public, immutable";
        }

        # =========================
        #  React static files (JS, CSS)
        # =========================
        location /static/ {
            alias C:/MyS_U4/MySpace/MyS_Front/build/static/;
            expires 1y;
            add_header Cache-Control "public, immutable";
        }

        # =========================
        #  React manifest, favicon, logos
        # =========================
        location ~* ^/(manifest\.json|favicon\.ico|logo192\.png|logo512\.png|robots\.txt)$ {
            root C:/MyS_U4/MySpace/MyS_Front/build;
            access_log off;
        }

        # =========================
        #  React index.html
        # =========================
        location / {
            root C:/MyS_U4/MySpace/MyS_Front/build;
            index index.html;
            try_files $uri /index.html;
        }

        # =========================
        #  Páginas de error
        # =========================
        error_page 500 502 503 504 /50x.html;
        location = /50x.html {
            root html;
        }
    }
}
