# =============================================
# NGINX CONFIG PARA DJANGO + REACT (Windows)
# Archivo: C:\nginx\conf\nginx.conf
# =============================================

# Número de procesos de trabajo (1 es suficiente en desarrollo)
worker_processes 1;

# =======================
# EVENTOS (config base)
# =======================
events {
    worker_connections 1024;
}

# =======================
# CONFIGURACIÓN HTTP
# =======================
http {
    include mime.types;
    default_type application/octet-stream;
    sendfile on;
    keepalive_timeout 65;

    # ==========================
    # BLOQUE PRINCIPAL DEL SERVER
    # ==========================
    server {
        listen 80;
        server_name localhost;

        # =====================
        # LOGS DE DEPURACIÓN
        # =====================
        access_log  C:/WebDeveloper/Proyecto_Programacion_Web/Web_U4/MyS_Back/nginx_access.log;
        error_log   C:/WebDeveloper/Proyecto_Programacion_Web/Web_U4/MyS_Back/nginx_error.log debug;

        # =====================
        # BACKEND DJANGO (Waitress)
        # =====================
        location /api/ {
            proxy_pass http://127.0.0.1:8001;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;

            # CORS
            add_header 'Access-Control-Allow-Origin' '*' always;
            add_header 'Access-Control-Allow-Methods' 'GET, POST, PUT, DELETE, OPTIONS' always;
            add_header 'Access-Control-Allow-Headers' 'Authorization, Content-Type' always;

            # Manejo de OPTIONS (preflight)
            if ($request_method = 'OPTIONS') {
                return 204;
            }
        }

        # =====================
        # ADMIN DE DJANGO
        # =====================
        location /admin/ {
            proxy_pass http://127.0.0.1:8001/admin/;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
        }

        # =====================
        # ARCHIVOS ESTÁTICOS DE DJANGO
        # =====================
        location /static/ {
            alias C:/WebDeveloper/Proyecto_Programacion_Web/Web_U4/MyS_Back/staticfiles/;
            expires 1y;
            add_header Cache-Control "public, immutable";
        }

        # =====================
        # ARCHIVOS ESTÁTICOS DE REACT
        # =====================
        location /front_static/ {
            alias C:/WebDeveloper/Proyecto_Programacion_Web/Web_U4/MyS_Back/static/;
            expires 1y;
            add_header Cache-Control "public, immutable";
        }

        # =====================
        # FRONTEND REACT (build)
        # =====================
        location / {
            root C:/WebDeveloper/Proyecto_Programacion_Web/Web_U4/MyS_Front/build;
            index index.html;
            try_files $uri $uri/ /index.html;
        }

        # =====================
        # PÁGINAS DE ERROR
        # =====================
        error_page 500 502 503 504 /50x.html;
        location = /50x.html {
            root html;
        }
    }
}
